# 1. INCIDENT HTTP 503
# ============================================================

# Compter les réponses 503 et déterminer le début et la fin de l'incident.

{
  "size": 0,
  "query": {
    "term": {
      "http.response.status_code": 503
    }
  },
  "aggs": {
    "debut_incident": {
      "min": {
        "field": "@timestamp"
      }
    },
    "fin_incident": {
      "max": {
        "field": "@timestamp"
      }
    }
  }
}
